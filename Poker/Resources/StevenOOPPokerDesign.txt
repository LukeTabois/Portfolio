- Suit (enum)

    + suit
    + image
    + toString (override)

- Deck

    + reset()

    + aceHigh
    + aceLow

- Player

    + hand (list<Card>)
    + handValue()
    + raise()
    + call()
    + fold()

- Table / Game

    + activePlayer (list<Players>)
    + deck
    + initialMinimumBet
    + currentMinimumBet
    + pot

    + deal()
    + takeMinimumBets()

    + turn(), 4th
    + river(), 5th
    + playerWithBestHand()
    + resetHand()
    + resetGame()
